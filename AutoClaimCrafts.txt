---------------------------------------------------------------- game access
local canId = type(setthreadidentity) == "function"
local idGame, idExec = 2, 7
if type(getthreadidentity) == "function" then
    local ok, id = pcall(getthreadidentity)
    if ok and type(id) == "number" then idExec = id end
end

local function gcall(fn, ...)
    if type(fn) ~= "function" then return false, "not a function" end
    if canId then pcall(setthreadidentity, idGame) end
    local r = table.pack(pcall(fn, ...))
    if canId then pcall(setthreadidentity, idExec) end
    return table.unpack(r, 1, r.n)
end
local function gval(fn, ...)
    local ok, a, b, c = gcall(fn, ...)
    if ok then return a, b, c end
    return nil
end
local function isfn(t, k) return type(t) == "table" and type(t[k]) == "function" end

local function req(path, optional)
    local ok, mod = gcall(function()
        local node = RS
        for seg in string.gmatch(path, "[^%.]+") do
            node = node:WaitForChild(seg, optional and 2 or 10) or error("missing " .. path)
        end
        return require(node)
    end)
    return ok and mod or nil
end

local Network, Directory, Instancing, FFlags, Forge
local function ensureModules()
    if Network and Directory then return true end
    Network = Network or req("Library.Client.Network")
    Directory = Directory or req("Library.Directory")
    Instancing = Instancing or req("Library.Client.InstancingCmds")
    FFlags = FFlags or req("Library.Client.FFlags", true)
    Forge = Forge or req("Library.Client.HalloweenGamesCmds.Forge", true)
    return Network ~= nil and Directory ~= nil
end

local function machineDir()
    local all = type(Directory) == "table" and rawget(Directory, "HalloweenCraftingMachines") or nil
    if all == nil and type(Directory) == "table" then
        local ok, v = pcall(function() return Directory.HalloweenCraftingMachines end)
        all = ok and v or nil
    end
    return type(all) == "table" and all[MACHINE_ID] or nil
end

local function flagValue(name)
    if not (name and type(FFlags) == "table" and type(FFlags.Keys) == "table" and type(FFlags.Get) == "function") then return nil end
    local key = FFlags.Keys[name]
    if key == nil then return nil end
    return gval(FFlags.Get, key)
end

local function ingredientNeed(dir, ing)
    local amount = tonumber(flagValue(ing.AmountFFlag)) or tonumber(ing.Amount) or 1
    local need = math.max(1, math.ceil(amount))
    if not ing.NoDiscount and type(dir.IngredientAmount) == "function" then
        local ok, v = pcall(dir.IngredientAmount, nil, need)
        if ok and type(v) == "number" and v == v then need = math.max(1, math.ceil(v)) end
    end
    return need
end

local function ingredientHave(dir, ing)
    local okC, have = gcall(function()
        local item = ing.Item:Clone()
        local n = item:CountAny()
        if dir.AnyVariantInputs and item:IsA("Pet") then
            for _, v in ipairs({ "Golden", "Rainbow" }) do
                local c = item:Clone()
                if c["CanBe" .. v](c) then
                    c["Set" .. v](c)
                    n += c:CountAny()
                end
            end
            if item:CanBeShiny() then
                for _, v in ipairs({ "Normal", "Golden", "Rainbow" }) do
                    local c = item:Clone()
                    local ok = v == "Normal" or (v == "Golden" and c:CanBeGolden()) or (v == "Rainbow" and c:CanBeRainbow())
                    if ok then
                        if v ~= "Normal" then c["Set" .. v](c) end
                        c:SetShiny(true)
                        n += c:CountAny()
                    end
                end
            end
        end
        return n
    end)
    return okC and tonumber(have) or 0
end

local onRecipes = function() end
local function readRecipes(dir)
    local out = {}
    if not dir or type(dir.Recipes) ~= "table" then return out end
    for i, r in ipairs(dir.Recipes) do
        if type(r) == "table" and r.Result and type(r.Ingredients) == "table" and r.Hidden ~= true then
            local name = tostring(gval(function() return r.Result:GetId() end) or ("Recipe " .. i))
            local open = r.EnabledFFlag == nil or flagValue(r.EnabledFFlag) ~= false
            local ready, missing = true, {}
            for _, ing in ipairs(r.Ingredients) do
                local need, have = ingredientNeed(dir, ing), ingredientHave(dir, ing)
                if have < need then
                    ready = false
                    missing[#missing + 1] = string.format("%s %d/%d", tostring(gval(function() return ing.Item:GetId() end) or "?"), have, need)
                end
            end
            out[#out + 1] = { index = i, name = name, open = open, ready = ready, missing = table.concat(missing, ", ") }
        end
    end
    lastRecipes = out
    pcall(onRecipes, out)
    return out
end

local function root()
    local c = LP.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end
local function tp(pos)
    local c, r = LP.Character, root()
    if not (c and r) then return end
    pcall(function()
        r.AssemblyLinearVelocity = Vector3.zero
        c:PivotTo(CFrame.new(pos) * (r.CFrame - r.CFrame.Position))
    end)
end
local function standPos()
    local ok, m = gcall(function() return Instancing.GetModel() end)
    if not (ok and typeof(m) == "Instance") then return nil end
    local inter = m:FindFirstChild("INTERACT")
    local machines = inter and inter:FindFirstChild("Machines")
    local mach = machines and machines:FindFirstChild(MACHINE_MODEL)
    if not mach then return nil end
    local pad = mach:FindFirstChild("Pad", true)
    if pad and pad:IsA("BasePart") then return pad.Position + Vector3.new(0, 3, 0) end
    return mach:GetPivot().Position + Vector3.new(0, 3, 6)
end

local function fmt(s)
    s = math.max(0, math.floor(s))
    if s >= 3600 then return string.format("%dh %dm", s // 3600, (s % 3600) // 60) end
    if s >= 60 then return string.format("%dm %ds", s // 60, s % 60) end
    return s .. "s"
end

local function pickedCount()
    local n = 0
    for _, v in pairs(picks) do if v then n += 1 end end
    return n
end

---------------------------------------------------------------- main logic
local busy = false
local function sweep(forceClaim, forceCraft)
    if busy then return end
    busy = true
    local atMachine, back = false, nil
    local ok, err = pcall(function()
        if not ensureModules() then claimNote = "Waiting for game..." return end
        local dir = machineDir()
        if not dir then claimNote = "Machine not found" return end
        if isfn(Forge, "CombineMode") and gval(Forge.CombineMode) == false then
            claimNote = "Combine-o-Matic is closed"
            return
        end

        local okQ, queue = gcall(Network.Invoke, "HalloweenCraftingMachine_GetQueue")
        if not okQ or type(queue) ~= "table" then claimNote = "Can't read queue" return end

        local queued, claims, nextIn = {}, {}, nil
        for _, e in ipairs(queue) do
            if type(e) == "table" and (e.MachineId == nil or e.MachineId == dir._id) and e.RecipeIndex then
                queued[e.RecipeIndex] = e
                local rem = tonumber(e.Remaining) or 0
                if rem <= 0 and e.UID then
                    claims[#claims + 1] = e
                elseif not nextIn or rem < nextIn then
                    nextIn = rem
                end
            end
        end

        local function goMachine()
            if atMachine or not gotoMachine then return false end
            local stand = standPos()
            if not stand then return false end
            local r = root()
            back = r and r.Position
            tp(stand)
            task.wait(0.8)
            atMachine = true
            return true
        end
        local function attempt(fn)
            if fn() then return true end
            if not atMachine and goMachine() then return fn() end
            return false
        end

        -- claim finished crafts
        if autoClaim or forceClaim then
            for _, e in ipairs(claims) do
                local done = attempt(function()
                    local ok2, res = gcall(Network.Invoke, "HalloweenCraftingMachine_Claim", e.UID)
                    return ok2 and res and true or false
                end)
                if done then
                    claimed += 1
                    queued[e.RecipeIndex] = nil
                end
                task.wait(0.5)
            end
        end
        claimNote = nextIn and ("Next in " .. fmt(nextIn)) or ((#queue == 0) and "Nothing crafting" or "All claimed")

        -- start picked crafts
        if (autoCraft or forceCraft) then
            if pickedCount() == 0 then craftNote = "Pick crafts below" return end
            local recipes = readRecipes(dir)
            local waiting, startedNow = nil, nil
            for i = #recipes, 1, -1 do
                local r = recipes[i]
                if picks[r.name] and r.open and not queued[r.index] then
                    if r.ready then
                        local msgOut
                        local done = attempt(function()
                            local ok2, res, msg = gcall(Network.Invoke, "HalloweenCraftingMachine_StartCraft", dir._id, r.index)
                            msgOut = msg or res
                            return ok2 and res and true or false
                        end)
                        if done then
                            started += 1
                            startedNow = r.name
                            queued[r.index] = true
                            task.wait(math.max(1, tonumber(dir.Cooldowns and dir.Cooldowns[1]) or 1) + 0.2)
                            -- pets were used: refresh readiness for the rest
                            local fresh = readRecipes(dir)
                            for j = 1, #fresh do recipes[j] = fresh[j] end
                        else
                            waiting = waiting or (r.name .. " refused: " .. tostring(msgOut))
                        end
                    else
                        waiting = waiting or (r.name .. " needs " .. r.missing)
                    end
                elseif picks[r.name] and queued[r.index] then
                    waiting = waiting or (r.name .. " is crafting")
                end
            end
            craftNote = startedNow and ("Started " .. startedNow) or (waiting or "Nothing to craft")
        end
    end)
    if atMachine and back then tp(back) end
    if not ok then claimNote = "Error (see F9)" warn("[AutoClaimCrafts] " .. tostring(err)) end
    busy = false
end

local function listHeight()
    local n = math.max(1, #lastRecipes)
    return math.min(MAXH, n * 28 + 10)
end

local function setOpen(v)
    open = v
    local info = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    TweenService:Create(arrow, info, { Rotation = open and 180 or 0 }):Play()
    TweenService:Create(ddStroke, info, { Color = open and ACCENT or OFF }):Play()
    if open then
        list.Visible = true
        TweenService:Create(list, info, { Size = UDim2.new(1, -24, 0, listHeight()) }):Play()
    else
        local t = TweenService:Create(list, info, { Size = UDim2.new(1, -24, 0, 0) })
        t.Completed:Connect(function() if not open then list.Visible = false end end)
        t:Play()
    end
end

local function updateDdText()
    local names = {}
    for _, r in ipairs(lastRecipes) do if picks[r.name] then names[#names + 1] = r.name end end
    if #names == 0 then ddText.Text = "Crafts: none selected"
    elseif #names == 1 then ddText.Text = names[1]
    else ddText.Text = string.format("Crafts: %d selected", #names) end
end

onRecipes = function(recipes)
    local key = ""
    for _, r in ipairs(recipes) do key ..= r.name .. "|" end
    if key ~= rowKey then
        rowKey = key
        for _, row in pairs(rows) do row.frame:Destroy() end
        rows = {}
        for i, r in ipairs(recipes) do
            local frame = mk("TextButton", {
                Size = UDim2.new(1, 0, 0, 26), BackgroundColor3 = PANEL2, BackgroundTransparency = 1,
                Text = "", AutoButtonColor = false, LayoutOrder = -i, ZIndex = 11,
            }, list)
            mk("UICorner", { CornerRadius = UDim.new(0, 6) }, frame)
            local box = mk("Frame", {
                Size = UDim2.fromOffset(14, 14), Position = UDim2.new(0, 6, 0.5, -7),
                BackgroundColor3 = picks[r.name] and ACCENT or OFF, ZIndex = 12,
            }, frame)
            mk("UICorner", { CornerRadius = UDim.new(0, 4) }, box)
            mk("TextLabel", {
                BackgroundTransparency = 1, Position = UDim2.fromOffset(28, 0), Size = UDim2.new(1, -46, 1, 0),
                Text = r.name, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd,
                Font = Enum.Font.Gotham, TextSize = 12, TextColor3 = TEXT, ZIndex = 12,
            }, frame)
            local dot = mk("Frame", {
                AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -6, 0.5, 0), Size = UDim2.fromOffset(8, 8),
                BackgroundColor3 = r.ready and GOOD or BAD, ZIndex = 12,
            }, frame)
            mk("UICorner", { CornerRadius = UDim.new(1, 0) }, dot)
            frame.MouseEnter:Connect(function() TweenService:Create(frame, TweenInfo.new(0.1), { BackgroundTransparency = 0.4 }):Play() end)
            frame.MouseLeave:Connect(function() TweenService:Create(frame, TweenInfo.new(0.1), { BackgroundTransparency = 1 }):Play() end)
            frame.MouseButton1Click:Connect(function()
                picks[r.name] = not picks[r.name] or nil
                TweenService:Create(box, TweenInfo.new(0.15), { BackgroundColor3 = picks[r.name] and ACCENT or OFF }):Play()
                updateDdText()
            end)
            rows[r.name] = { frame = frame, box = box, dot = dot }
        end
        if open then TweenService:Create(list, TweenInfo.new(0.2), { Size = UDim2.new(1, -24, 0, listHeight()) }):Play() end
    else
        for _, r in ipairs(recipes) do
            local row = rows[r.name]
            if row then row.dot.BackgroundColor3 = r.ready and GOOD or BAD end
        end
    end
    updateDdText()
end

task.spawn(function()
    while alive do
